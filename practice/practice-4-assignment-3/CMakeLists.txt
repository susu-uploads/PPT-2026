cmake_minimum_required(VERSION 3.20)
project(PPTPractice3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenMP REQUIRED COMPONENTS CXX)

add_executable(task-6 "omp-loop-scheduling.cpp")
target_compile_options(task-6 PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(task-6 PRIVATE OpenMP::OpenMP_CXX)

add_executable(task-8 "omp-matrix-multiply.cpp")
target_compile_options(task-8 PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(task-8 PRIVATE OpenMP::OpenMP_CXX)

add_executable(task-9 "omp-sections.cpp")
target_compile_options(task-9 PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(task-9 PRIVATE OpenMP::OpenMP_CXX)

include("../../cmake/PracticeTests.cmake")
if(BUILD_TESTING)
  add_program_test(sum-k1-n5 task-6 INPUT "1 5\n" CHECK sum 1 5)
  add_program_test(sum-k2-n4 task-6 INPUT "2 4\n" CHECK sum 2 4)
  add_program_test(sum-k3-n2 task-6 INPUT "3 2\n" CHECK sum 3 2)
  add_program_test(sum-k4-n0 task-6 INPUT "4 0\n" CHECK sum 4 0)
  add_program_test(sum-k4-n10000 task-6 INPUT "4 10000\n" CHECK sum 4 10000)
  foreach(schedule IN ITEMS static static,1 static,2 dynamic dynamic,2 guided guided,2)
    string(REPLACE "," "-" name "${schedule}")
    add_program_test(schedule-${name} task-6 ARGS --schedule "${schedule}" CHECK schedule 4 10)
  endforeach()
  foreach(n IN ITEMS 1 2 3 10)
    matrix_case(${n} input expected)
    add_program_test(matrix-n${n} task-8 INPUT "${input}" CHECK matrix ${expected})
  endforeach()
  add_program_test(matrix-example task-8 INPUT "2\n1 1 4 4\n5 5 3 3\n" CHECK matrix 8 8 32 32)
  foreach(k IN ITEMS 1 2 3 4)
    add_program_test(sections-k${k} task-9 INPUT "${k}\n" CHECK sections ${k})
  endforeach()
  add_program_test(invalid-matrix-size task-8 INPUT "11\n" EXIT_CODE 2
    CHECK error "Размер матриц: 1..10.")
  add_program_test(incomplete-matrix task-8 INPUT "2\n1 2\n" EXIT_CODE 2
    CHECK error "Неполная матрица A.")
endif()
